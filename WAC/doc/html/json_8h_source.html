33inline std::wstring jsonEscape(const std::wstring& s) {
34 std::wstring r;
35 r.reserve(s.size() + 8);
36 for (wchar_t c : s) {
37 switch (c) {
38 case L'\\': r += L"\\\\"; break;
39 case L'"': r += L"\\\""; break;
40 case L'\b': r += L"\\b"; break;
41 case L'\f': r += L"\\f"; break;
42 case L'\n': r += L"\\n"; break;
43 case L'\r': r += L"\\r"; break;
44 case L'\t': r += L"\\t"; break;
45 default:
46 if (c < 0x20) { // other control characters
47 static const wchar_t* h = L"0123456789abcdef";
48 r += L"\\u00"; r += h[(c >> 4) & 0xF]; r += h[c & 0xF];
49 }
50 else r += c;
51 }
52 }
53 return r;
54}

244 if (c >= L'0' && c <= L'9') v |= (unsigned)(c - L'0');

251 bool string(std::wstring& out) {
252 ++pos; // the opening quote
253 out.clear();
254 while (true) {
255 if (pos >= s.size()) return fail(L"unterminated string");

265 case L'/': out += L'/'; break;

306 ++pos;
307 blank();

309 while (true) {

311 if (object) {
312 blank();

314 if (!string(key)) return false;

317 ++pos;
318 }

322 blank();
323 if (pos >= s.size()) return fail(object ? L"unterminated object" : L"unterminated array");
324 if (s[pos] == L',') { ++pos; continue; }
325 if (s[pos] == (object ? L'}' : L']')) { ++pos; --depth; return true; }
326 return fail(L"expected ',' or a closing bracket");

328 }
329 if (c == L'"') { out = Json(Kind::Str); return string(out.scalar_); }
330 if (c == L't') { if (!literal(L"true")) return false; out = Json(Kind::Bool); out.scalar_ = L"true"; return true; }
331 if (c == L'f') { if (!literal(L"false")) return false; out = Json(Kind::Bool); out.scalar_ = L"false"; return true; }
332 if (c == L'n') { if (!literal(L"null")) return false; out = Json(Kind::Null); return true; }
