31inline std::wstring jsonEscape(const std::wstring& s) {
32 std::wstring r;
33 r.reserve(s.size() + 8);
34 for (wchar_t c : s) {
35 switch (c) {
36 case L'\\': r += L"\\\\"; break;
37 case L'"': r += L"\\\""; break;
38 case L'\b': r += L"\\b"; break;
39 case L'\f': r += L"\\f"; break;
40 case L'\n': r += L"\\n"; break;
41 case L'\r': r += L"\\r"; break;
42 case L'\t': r += L"\\t"; break;
43 default:
44 if (c < 0x20) { // other control characters
45 static const wchar_t* h = L"0123456789abcdef";
46 r += L"\\u00"; r += h[(c >> 4) & 0xF]; r += h[c & 0xF];
47 }
48 else r += c;
49 }
50 }
51 return r;
52}
