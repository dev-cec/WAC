32inline std::wstring jsonEscape(const std::wstring& s) {
33 std::wstring r;
34 r.reserve(s.size() + 8);
35 for (wchar_t c : s) {
36 switch (c) {
37 case L'\\': r += L"\\\\"; break;
38 case L'"': r += L"\\\""; break;
39 case L'\b': r += L"\\b"; break;
40 case L'\f': r += L"\\f"; break;
41 case L'\n': r += L"\\n"; break;
42 case L'\r': r += L"\\r"; break;
43 case L'\t': r += L"\\t"; break;
44 default:
45 if (c < 0x20) { // other control characters
46 static const wchar_t* h = L"0123456789abcdef";
47 r += L"\\u00"; r += h[(c >> 4) & 0xF]; r += h[c & 0xF];
48 }
49 else r += c;
50 }
51 }
52 return r;
53}
