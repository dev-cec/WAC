1/* json.h — sérialiseur JSON centralisé pour WAC.
2 *
3 * POURQUOI. Chaque artefact fabriquait son JSON à la main par concaténation de
4 * chaînes. Conséquences constatées en test : échappement oublié ou incohérent
5 * (ScheduledTasks, mounted_device), virgule finale invalide (Sessions), nombres
6 * émis comme chaînes, indentation gérée à la main. Ce writer rend ces erreurs
7 * structurellement impossibles.
8 *
9 * PRINCIPE. Les valeurs sont stockées BRUTES en mémoire (vrais chemins,
10 * utilisables tels quels pour les I/O fichier) ; l'échappement n'a lieu QU'ICI,
11 * à la sérialisation, et une seule fois.
12 *
13 * Autonome (aucune dépendance au reste de WAC) : testable sous Linux.
14 *
15 * Exemple :
16 * Json o = Json::obj();
17 * o.add(L"SessionId", Json::num(sessionId)); // nombre, sans guillemets
18 * o.add(L"LogonName", Json::str(logonName)); // échappé automatiquement
19 * o.add(L"Path", Json::str(cheminBrut)); // chemin brut, échappé ici
20 * return o.dump(1);
21 */

28inline std::wstring jsonEscape(const std::wstring& s) {
29 std::wstring r;
30 r.reserve(s.size() + 8);
31 for (wchar_t c : s) {
32 switch (c) {
33 case L'\\': r += L"\\\\"; break;
34 case L'"': r += L"\\\""; break;
35 case L'\b': r += L"\\b"; break;
36 case L'\f': r += L"\\f"; break;
37 case L'\n': r += L"\\n"; break;
38 case L'\r': r += L"\\r"; break;
39 case L'\t': r += L"\\t"; break;
40 default:
41 if (c < 0x20) { // autres caractères de contrôle
42 static const wchar_t* h = L"0123456789abcdef";
43 r += L"\\u00"; r += h[(c >> 4) & 0xF]; r += h[c & 0xF];
44 }
45 else r += c;
46 }
47 }
48 return r;
49}
