46inline std::wstring extendedPath(const std::wstring& path) {
47 // MAX_PATH less the 12 characters CreateDirectoryW keeps for an 8.3 name.
48 const size_t LIMIT = MAX_PATH - 12;
49 if (path.empty() || path.compare(0, 4, L"\\\\?\\") == 0) return path;
50 const DWORD size = GetFullPathNameW(path.c_str(), 0, nullptr, nullptr);
51 if (size == 0) return path;
52 std::wstring full(size, L'\0');
53 const DWORD written = GetFullPathNameW(path.c_str(), size, full.data(), nullptr);
54 if (written == 0 || written >= size) return path;
55 full.resize(written);
56 if (full.size() < LIMIT) return path;
57 if (full.compare(0, 2, L"\\\\") == 0) return L"\\\\?\\UNC\\" + full.substr(2);
58 return L"\\\\?\\" + full;
59}

64inline bool createFolders(const std::wstring& folder) {
65 const std::wstring path = extendedPath(folder);
66 const DWORD attributes = GetFileAttributesW(path.c_str());
67 if (attributes != INVALID_FILE_ATTRIBUTES) return (attributes & FILE_ATTRIBUTE_DIRECTORY) != 0;
68 const size_t cut = path.find_last_of(L"\\/");
69 // The parent first; a root ("C:", "\\?\C:") has none to create.
70 if (cut != std::wstring::npos && cut > 0 && path[cut - 1] != L':' && path[cut - 1] != L'?')
71 createFolders(path.substr(0, cut));
72 return CreateDirectoryW(path.c_str(), nullptr) || GetLastError() == ERROR_ALREADY_EXISTS;
73}

90inline bool fileSizeOf(const std::wstring& file, uint64_t& size) {
91 WIN32_FILE_ATTRIBUTE_DATA data = {};
92 if (!GetFileAttributesExW(extendedPath(file).c_str(), GetFileExInfoStandard, &data)) return false;
93 size = ((uint64_t)data.nFileSizeHigh << 32) | data.nFileSizeLow;
94 return true;

125inline bool forEachFileBelow(const std::wstring& folder, const std::function<void(const std::wstring&)>& visit,
126 const std::function<bool(const std::wstring&)>& enter = nullptr) {
127 const std::wstring root = extendedPath(folder);
128 std::vector<std::wstring> pending{ L"" };
129 bool first = true;
130 while (!pending.empty()) {
131 const std::wstring relative = pending.back();
132 pending.pop_back();
133 // The folder joined to its relative part, and only then extended: the
134 // sum may pass the limit where neither did.
135 const std::wstring here = extendedPath(relative.empty() ? root : root + L"\\" + relative);
136 WIN32_FIND_DATAW found = {};
137 const HANDLE h = FindFirstFileExW((here + L"\\*").c_str(), FindExInfoBasic, &found,
138 FindExSearchNameMatch, nullptr, FIND_FIRST_EX_LARGE_FETCH);
139 if (h == INVALID_HANDLE_VALUE) {
140 if (first) return false;
141 continue;
142 }
143 first = false;
144 std::vector<std::wstring> subfolders;
145 do {
146 const std::wstring name = found.cFileName;
147 if (name == L"." || name == L"..") continue;
148 const std::wstring child = relative.empty() ? name : relative + L"\\" + name;
149 // A junction or a symbolic link is not followed: it can loop.
150 if (found.dwFileAttributes & FILE_ATTRIBUTE_REPARSE_POINT) continue;
151 if (found.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
152 if (!enter || enter(child)) subfolders.push_back(child);
153 }
154 else visit(child);
155 } while (FindNextFileW(h, &found));
156 FindClose(h);
157 // Pushed in reverse: the first listed is walked first.
158 pending.insert(pending.end(), subfolders.rbegin(), subfolders.rend());
159 }
160 return true;
161}

166inline bool removeFolderTree(const std::wstring& folder) {
167 const std::wstring root = extendedPath(folder);
168 if (GetFileAttributesW(root.c_str()) == INVALID_FILE_ATTRIBUTES) return true;
169 std::vector<std::wstring> folders{ L"" }; // in the order found: parents first
170 forEachFileBelow(folder, [&](const std::wstring& relative) {
171 const std::wstring file = extendedPath(root + L"\\" + relative);
172 SetFileAttributesW(file.c_str(), FILE_ATTRIBUTE_NORMAL); // a read-only exhibit too
173 DeleteFileW(file.c_str());
174 });
175 // The folders, listed parents first, removed children first.
176 for (size_t i = 0; i < folders.size(); ++i) {
177 const std::wstring here = extendedPath(folders[i].empty() ? root : root + L"\\" + folders[i]);
178 WIN32_FIND_DATAW found = {};
179 const HANDLE h = FindFirstFileExW((here + L"\\*").c_str(), FindExInfoBasic, &found,
180 FindExSearchNameMatch, nullptr, 0);
181 if (h == INVALID_HANDLE_VALUE) continue;
182 do {
183 const std::wstring name = found.cFileName;
184 if (name == L"." || name == L"..") continue;
185 // A junction is removed as a folder, never entered: its target is not ours.
186 if ((found.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) && !(found.dwFileAttributes & FILE_ATTRIBUTE_REPARSE_POINT))
187 folders.push_back(folders[i].empty() ? name : folders[i] + L"\\" + name);
188 else if (found.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY)
189 RemoveDirectoryW(extendedPath(here + L"\\" + name).c_str());
190 } while (FindNextFileW(h, &found));
191 FindClose(h);
192 }
193 for (size_t i = folders.size(); i-- > 0;)
194 RemoveDirectoryW(extendedPath(folders[i].empty() ? root : root + L"\\" + folders[i]).c_str());
195 return GetFileAttributesW(root.c_str()) == INVALID_FILE_ATTRIBUTES;
196}
