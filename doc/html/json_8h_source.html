35inline std::wstring jsonEscape(const std::wstring& s) {
36 std::wstring r;
37 r.reserve(s.size() + 8);
38 for (wchar_t c : s) {
39 switch (c) {
40 case L'\\': r += L"\\\\"; break;
41 case L'"': r += L"\\\""; break;
42 case L'\b': r += L"\\b"; break;
43 case L'\f': r += L"\\f"; break;
44 case L'\n': r += L"\\n"; break;
45 case L'\r': r += L"\\r"; break;
46 case L'\t': r += L"\\t"; break;
47 default:
48 if (c < 0x20) { // other control characters
49 static const wchar_t* h = L"0123456789abcdef";
50 r += L"\\u00"; r += h[(c >> 4) & 0xF]; r += h[c & 0xF];
51 }
52 else r += c;
53 }
54 }
55 return r;
56}
