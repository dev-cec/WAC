36inline std::wstring jsonEscape(const std::wstring& s) {
37 static const wchar_t* hex = L"0123456789abcdef";
38 std::wstring r;
39 r.reserve(s.size() + 8);
40 for (size_t i = 0; i < s.size(); ++i) {
41 const wchar_t c = s[i];
42 /* A lone surrogate, which an NTFS name may hold (measured: User Access
43 Logging's databases on Server 2016), has no UTF-8 form: converted, it
44 would become U+FFFD, two names alike. Escaped, it stays itself. */
45 const bool high = c >= 0xD800 && c <= 0xDBFF, low = c >= 0xDC00 && c <= 0xDFFF;
46 if (high && i + 1 < s.size() && s[i + 1] >= 0xDC00 && s[i + 1] <= 0xDFFF) {
47 r += c; r += s[++i];
48 continue;
49 }
50 if (high || low) {
51 r += L"\\u";
52 for (int shift = 12; shift >= 0; shift -= 4) r += hex[(c >> shift) & 0xF];
53 continue;
54 }
55 switch (c) {
56 case L'\\': r += L"\\\\"; break;
57 case L'"': r += L"\\\""; break;
58 case L'\b': r += L"\\b"; break;
59 case L'\f': r += L"\\f"; break;
60 case L'\n': r += L"\\n"; break;
61 case L'\r': r += L"\\r"; break;
62 case L'\t': r += L"\\t"; break;
63 default:
64 if (c < 0x20) { // other control characters
65 r += L"\\u00"; r += hex[(c >> 4) & 0xF]; r += hex[c & 0xF];
66 }
67 else r += c;
68 }
69 }
70 return r;
71}

266 bool hex4(unsigned& v) {
267 if (s.size() - pos < 4) return fail(L"truncated \\u escape");
268 v = 0;
269 for (int k = 0; k < 4; ++k, ++pos) {
270 const wchar_t c = s[pos];
271 v <<= 4;
272 if (c >= L'0' && c <= L'9') v |= (unsigned)(c - L'0');
273 else if (c >= L'a' && c <= L'f') v |= (unsigned)(c - L'a' + 10);
274 else if (c >= L'A' && c <= L'F') v |= (unsigned)(c - L'A' + 10);
275 else return fail(L"invalid \\u escape");
276 }
277 return true;
278 }
279 bool string(std::wstring& out) {
280 ++pos; // the opening quote
281 out.clear();
282 while (true) {
283 if (pos >= s.size()) return fail(L"unterminated string");
284 const wchar_t c = s[pos++];
285 if (c == L'"') return true;
286 if (c < 0x20) return fail(L"control character in a string");
287 if (c != L'\\') { out += c; continue; }
288 if (pos >= s.size()) return fail(L"truncated escape");
289 const wchar_t e = s[pos++];
290 switch (e) {
291 case L'"': out += L'"'; break;
292 case L'\\': out += L'\\'; break;
293 case L'/': out += L'/'; break;
294 case L'b': out += L'\b'; break;
295 case L'f': out += L'\f'; break;
296 case L'n': out += L'\n'; break;
297 case L'r': out += L'\r'; break;
298 case L't': out += L'\t'; break;
299 case L'u': { unsigned v = 0; if (!hex4(v)) return false; out += (wchar_t)v; break; }
300 default: return fail(L"invalid escape");
301 }
302 }
303 }
304 bool number(Json& out) {
305 const size_t start = pos;
306 if (pos < s.size() && s[pos] == L'-') ++pos;
307 if (pos >= s.size()) return fail(L"truncated number");
308 if (s[pos] == L'0') ++pos;
309 else if (s[pos] >= L'1' && s[pos] <= L'9') { while (pos < s.size() && s[pos] >= L'0' && s[pos] <= L'9') ++pos; }
310 else return fail(L"invalid number");
311 if (pos < s.size() && s[pos] == L'.') {
312 ++pos;
313 if (pos >= s.size() || s[pos] < L'0' || s[pos] > L'9') return fail(L"invalid fraction");
314 while (pos < s.size() && s[pos] >= L'0' && s[pos] <= L'9') ++pos;
315 }
316 if (pos < s.size() && (s[pos] == L'e' || s[pos] == L'E')) {
317 ++pos;
318 if (pos < s.size() && (s[pos] == L'+' || s[pos] == L'-')) ++pos;
319 if (pos >= s.size() || s[pos] < L'0' || s[pos] > L'9') return fail(L"invalid exponent");
320 while (pos < s.size() && s[pos] >= L'0' && s[pos] <= L'9') ++pos;
321 }
322 out = Json(Kind::Num);
323 out.scalar_ = s.substr(start, pos - start);
324 return true;
325 }
326 bool value(Json& out) {
327 blank();
328 if (pos >= s.size()) return fail(L"missing value");
329 const wchar_t c = s[pos];
330 if (c == L'{' || c == L'[') {
331 if (++depth > MAX_DEPTH) return fail(L"nesting too deep");
332 const bool object = (c == L'{');
333 out = Json(object ? Kind::Obj : Kind::Arr);
334 ++pos;
335 blank();
336 if (pos < s.size() && s[pos] == (object ? L'}' : L']')) { ++pos; --depth; return true; }
337 while (true) {
338 std::wstring key;
339 if (object) {
340 blank();
341 if (pos >= s.size() || s[pos] != L'"') return fail(L"expected a key");
342 if (!string(key)) return false;
343 blank();
344 if (pos >= s.size() || s[pos] != L':') return fail(L"expected ':'");
345 ++pos;
346 }
347 Json member(Kind::Null);
348 if (!value(member)) return false;
349 out.items_.emplace_back(std::move(key), std::move(member));
350 blank();
351 if (pos >= s.size()) return fail(object ? L"unterminated object" : L"unterminated array");
352 if (s[pos] == L',') { ++pos; continue; }
353 if (s[pos] == (object ? L'}' : L']')) { ++pos; --depth; return true; }
354 return fail(L"expected ',' or a closing bracket");
355 }
356 }
357 if (c == L'"') { out = Json(Kind::Str); return string(out.scalar_); }
358 if (c == L't') { if (!literal(L"true")) return false; out = Json(Kind::Bool); out.scalar_ = L"true"; return true; }
359 if (c == L'f') { if (!literal(L"false")) return false; out = Json(Kind::Bool); out.scalar_ = L"false"; return true; }
360 if (c == L'n') { if (!literal(L"null")) return false; out = Json(Kind::Null); return true; }
361 if (c == L'-' || (c >= L'0' && c <= L'9')) return number(out);
362 return fail(L"unexpected character");
363 }
